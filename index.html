"'" + s : s; } function hoja_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sh = ss.getSheetByName(HOJA) || ss.insertSheet(HOJA); if (sh.getLastRow() === 0) { sh.appendRow(ENCABEZADOS); sh.setFrozenRows(1); sh.getRange(1, 1, 1, ENCABEZADOS.length).setFontWeight("bold"); } return sh; } function pesos_(n) { return "$" + String(Math.round(Number(n) || 0)).replace(/\B(?=(\d{3})+(?!\d))/g, "."); } function avisar_(p, carpetaUrl, fotos) { try { const destinos = AVISAR_A.filter(function (c) { return /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(c); }); if (!destinos.length) return; const lineas = p.i.map(function (x) { return "- " + limpiar_(x[1], 160) + ": " + pesos_(x[2]); }); const cel = limpiar_(p.c, 10); const cuerpo = [ "Llegó una cotización nueva por la página.", "", "Cliente: " + limpiar_(p.n, 60), "Celular: " + cel + (cel ? } // La página envía aquí la cotización (POST). function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const d = JSON.parse(e.postData.contents); const m = String(d.ref || "").match(/^LS1\.([A-Za-z0-9_-]+)$/); if (String(p.token || "").normalize("NFC") !== CLAVE.normalize("NFC")) return json_({ ok: false, error: "clave" }); if (p.action === "list") { const sh = hoja_(); const ultima = sh.getLastRow(); const filas = []; }); } return json_({ ok: true, rows: filas }); } return json_({ ok: true }); }
